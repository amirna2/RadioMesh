123template <typename T>
124std::vector<byte> numberToBytes(T number)
125{
126 std::vector<byte> bytes(sizeof(T));
127 for (size_t i = 0; i < sizeof(T); ++i) {
128 bytes[i] = static_cast<byte>((number >> (8 * (sizeof(T) - 1 - i))) & 0xFF);
129 }
130 return bytes;
131}

138template <typename T>
139T bytesToNumber(const std::vector<byte>& bytes)
140{
141 T number = 0;
142 size_t limit = std::min(bytes.size(), sizeof(T));
143 for (size_t i = 0; i < limit; ++i) {
144 number = (number << 8) | bytes[i];
145 }
146 return number;
147}
