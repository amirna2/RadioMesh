112template <typename T>
113std::vector<byte> numberToBytes(T number)
114{
115 std::vector<byte> bytes(sizeof(T));
116 for (size_t i = 0; i < sizeof(T); ++i) {
117 bytes[i] = static_cast<byte>((number >> (8 * (sizeof(T) - 1 - i))) & 0xFF);
118 }
119 return bytes;
120}

127template <typename T>
128T bytesToNumber(const std::vector<byte>& bytes)
129{
130 T number = 0;
131 size_t limit = std::min(bytes.size(), sizeof(T));
132 for (size_t i = 0; i < limit; ++i) {
133 number = (number << 8) | bytes[i];
134 }
135 return number;
136}
