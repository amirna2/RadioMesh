122template <typename T>
123std::vector<byte> numberToBytes(T number)
124{
125 std::vector<byte> bytes(sizeof(T));
126 for (size_t i = 0; i < sizeof(T); ++i) {
127 bytes[i] = static_cast<byte>((number >> (8 * (sizeof(T) - 1 - i))) & 0xFF);
128 }
129 return bytes;
130}

137template <typename T>
138T bytesToNumber(const std::vector<byte>& bytes)
139{
140 T number = 0;
141 size_t limit = std::min(bytes.size(), sizeof(T));
142 for (size_t i = 0; i < limit; ++i) {
143 number = (number << 8) | bytes[i];
144 }
145 return number;
146}
